var bot = BotManager.getCurrentBot();

var File = java.io.File;
var FileOutputStream = java.io.FileOutputStream;
var OutputStreamWriter = java.io.OutputStreamWriter;
var BufferedWriter = java.io.BufferedWriter;
var SimpleDateFormat = java.text.SimpleDateFormat;
var Date = java.util.Date;
var Locale = java.util.Locale;

var CONFIG = {
  ROOT_DIR: "/storage/emulated/0/Download/wiki/kakao_wiki",
  RAW_DIR: "raw/방이름",
  DEBUG_DIR: "debug",
  TARGET_ROOM: "방이름",
  FILE_PREFIX: "kakao_방이름_",
  DEBUG_PREFIX: "debug_"
};

function ensureDir(path) {
  var f = new File(path);
  if (!f.exists()) {
    f.mkdirs();
  }
}

function rootDir() {
  return CONFIG.ROOT_DIR;
}

function rawDir() {
  return rootDir() + "/" + CONFIG.RAW_DIR;
}

function debugDir() {
  return rootDir() + "/" + CONFIG.DEBUG_DIR;
}

function fileExists(path) {
  try {
    return new File(path).exists();
  } catch (e) {
    return false;
  }
}

function writeText(path, text, append) {
  ensureDir(new File(path).getParent());
  var writer = null;
  try {
    writer = new BufferedWriter(new OutputStreamWriter(new FileOutputStream(path, !!append), "UTF-8"));
    writer.write(String(text));
    writer.flush();
  } finally {
    try { if (writer) writer.close(); } catch (e) {}
  }
}

function nowKey() {
  return new SimpleDateFormat("yyMMdd", Locale.getDefault()).format(new Date());
}

function nowStamp() {
  return new SimpleDateFormat("yyyy-MM-dd HH:mm:ss", Locale.getDefault()).format(new Date());
}

function normalizeText(v) {
  if (v === null || v === undefined) return "";
  return String(v).replace(/\r/g, "").replace(/\n+/g, " ").replace(/\s+/g, " ").trim();
}

function getRoomName(msg) {
  if (!msg) return "";
  try {
    if (msg.roomName) return normalizeText(msg.roomName);
  } catch (e) {}
  try {
    if (msg.chatRoomName) return normalizeText(msg.chatRoomName);
  } catch (e) {}
  try {
    if (msg.room && msg.room.name) return normalizeText(msg.room.name);
  } catch (e) {}
  try {
    if (msg.chatroom && msg.chatroom.name) return normalizeText(msg.chatroom.name);
  } catch (e) {}
  try {
    if (msg.room) return normalizeText(msg.room);
  } catch (e) {}
  return "";
}

function getAuthorName(msg) {
  try {
    if (msg && msg.author && msg.author.name) return normalizeText(msg.author.name);
  } catch (e) {}
  return "unknown";
}

function getContent(msg) {
  try {
    return normalizeText(msg && msg.content);
  } catch (e) {
    return "";
  }
}

function sameRoom(room) {
  var target = normalizeText(CONFIG.TARGET_ROOM);
  var current = normalizeText(room);
  if (!current) return false;
  if (current === target) return true;
  if (current.indexOf(target) !== -1) return true;
  return false;
}

function rawPath() {
  return rawDir() + "/" + CONFIG.FILE_PREFIX + nowKey() + ".md";
}

function debugPath() {
  return debugDir() + "/" + CONFIG.DEBUG_PREFIX + nowKey() + ".md";
}

function ensureHeaders() {
  ensureDir(rootDir());
  ensureDir(rawDir());
  ensureDir(debugDir());
}

function appendRawLine(msg) {
  var room = getRoomName(msg);
  if (!sameRoom(room)) return;

  var content = getContent(msg);
  var author = getAuthorName(msg);
  var t = nowStamp();
  var path = rawPath();

  if (!fileExists(path)) {
    var header = [];
    header.push("# kakao_ez raw log");
    header.push("");
    header.push("- date: " + t.substring(0, 10));
    header.push("- room: " + room);
    header.push("- source: MessengerBotR");
    header.push("");
    writeText(path, header.join("\n") + "\n", false);
  }

  var line = "[" + t.substring(11, 16) + "]" + author.replace(/\n/g, " ").replace(/\r/g, " ") + ":" + content.replace(/\n/g, " ").replace(/\r/g, " ");
  writeText(path, line + "\n", true);
}

function appendDebugLine(msg) {
  var room = getRoomName(msg);
  var content = getContent(msg);
  var author = getAuthorName(msg);
  var path = debugPath();

  if (!fileExists(path)) {
    writeText(path, "# kakao_ez debug log\n\n", false);
  }

  var keys = [];
  try {
    for (var k in msg) {
      if (keys.length >= 25) break;
      keys.push(String(k));
    }
  } catch (e) {}
  
var line = [];
  line.push("- ts=" + nowStamp() + "");
  line.push("room=" + room.replace(//g, "'") + "");
  line.push("author=" + author.replace(//g, "'") + "");
  line.push("content=" + content.replace(//g, "'") + "");
  if (keys.length > 0) line.push("keys=" + keys.join(",") + "");
  writeText(path, line.join(" | ") + "\n", true);
}

function onMessage(msg) {
  try {
    ensureHeaders();
    appendDebugLine(msg);
    appendRawLine(msg);
  } catch (e) {
    try {
      ensureHeaders();
      writeText(debugPath(), "- error=" + String(e).replace(//g, "'") + "\n", true);
    } catch (e2) {}
  }
}

function onCommand(msg) {
  try {
    var cmd = String(msg.command || "").toLowerCase();
    var args = msg.args || [];
    if (cmd === "kakaoez") {
      if (args.length > 0 && String(args[0]).toLowerCase() === "status") {
        msg.reply("logger running\n- raw: " + rawPath() + "\n- debug: " + debugPath());
      } else {
        msg.reply("@kakaoez status");
      }
    }
  } catch (e) {}
}

function onCreate(savedInstanceState, activity) {
  ensureHeaders();
  var textView = new android.widget.TextView(activity);
  textView.setText("kakao_ez logger ready");
  textView.setTextColor(android.graphics.Color.DKGRAY);
  activity.setContentView(textView);
}

function onStart(activity) {}
function onResume(activity) {}
function onPause(activity) {}
function onStop(activity) {}
function onRestart(activity) {}
function onDestroy(activity) {}
function onBackPressed(activity) {}

bot.addListener(Event.MESSAGE, onMessage);
bot.setCommandPrefix("@");
bot.addListener(Event.COMMAND, onCommand);
bot.addListener(Event.Activity.CREATE, onCreate);
bot.addListener(Event.Activity.START, onStart);
bot.addListener(Event.Activity.RESUME, onResume);
bot.addListener(Event.Activity.PAUSE, onPause);
bot.addListener(Event.Activity.STOP, onStop);
bot.addListener(Event.Activity.RESTART, onRestart);
bot.addListener(Event.Activity.DESTROY, onDestroy);
bot.addListener(Event.Activity.BACK_PRESSED, onBackPressed);
